# ---- Robust GET with retries/timeouts/backoff --------------------------------
retry_GET <- function(url, query_list, max_attempts = 6, pause_secs = 2, req_timeout = 120) {
  attempt <- 1
  repeat {
    res <- tryCatch(
      GET(url, query = query_list, timeout(req_timeout)),
      error = function(e) e
    )
    if (inherits(res, "error")) {
      if (attempt >= max_attempts) stop(res)
      message(sprintf("Retry %d - %s; sleeping %d sec...", attempt, conditionMessage(res), pause_secs))
      Sys.sleep(pause_secs); pause_secs <- min(pause_secs * 2, 60); attempt <- attempt + 1
      next
    }
    if (status_code(res) == 503) {
      if (attempt >= max_attempts) stop("Max retry attempts reached due to 503 error.")
      message(sprintf("Retry %d - 503 Error; sleeping %d sec...", attempt, pause_secs))
      Sys.sleep(pause_secs); pause_secs <- min(pause_secs * 2, 60); attempt <- attempt + 1
      next
    }
    stop_for_status(res)
    return(res)
  }
}

fetch_page <- function(uuid, filter_list, offset, size = 5000, polite_sleep = 0.2) {
  full_url <- sprintf(base_url_template, uuid)
  query_params <- c(list("size" = size, "offset" = offset), filter_list)
  res <- retry_GET(full_url, query_params)
  raw <- content(res, as = "text", encoding = "UTF-8")
  dat <- fromJSON(raw, flatten = TRUE)
  if (is.null(dat) || length(dat) == 0) dat <- tibble()
  Sys.sleep(polite_sleep)
  as_tibble(dat)
}

# Sequential paginator with safety limits and a CPT validation check to avoid
# pulling the full dataset when a filter is ignored by the API.
fetch_all_for_filter <- function(uuid, filter_list, cpt_code, page_size = 5000, max_pages = 20) {
  out <- list(); offset <- 0L; page_idx <- 1L
  code_candidates <- c("HCPCS_CD", "HCPCS_Cd", "Rndrng_Prvdr_HCPCS_Cd")

  repeat {
    page <- fetch_page(uuid, filter_list, offset, size = page_size)
    if (nrow(page) == 0) break

    if (page_idx == 1) {
      code_col <- intersect(code_candidates, names(page))[1]
      if (!is.na(code_col)) {
        if (!cpt_code %in% unique(page[[code_col]])) {
          message(sprintf("    ⚠ First page missing CPT %s in column %s; abandoning this filter", cpt_code, code_col))
          return(tibble())
        }
      }
    }

    out[[length(out) + 1]] <- page
    if (nrow(page) < page_size) break
    if (page_idx >= max_pages) {
      message(sprintf("    ⚠ Hit max_pages=%d for this filter; stopping pagination", max_pages))
      break
    }

    offset <- offset + page_size
    page_idx <- page_idx + 1L
  }

  if (length(out) == 0) tibble() else bind_rows(out)
}

dedupe_year <- function(df) {
  if (nrow(df) == 0) return(df)
  key_cols <- intersect(c("HCPCS_CD", "CARRIER_NUM", "PRICING_LOCALITY_CD", "TYPE_OF_SERVICE_CD", "PLACE_OF_SERVICE_CD"), names(df))
  if (length(key_cols) >= 2) distinct(df, across(all_of(key_cols)), .keep_all = TRUE) else distinct(df)
}

get_year_data <- function(year, uuid, cpt_code, cpt_label) {
  message(sprintf("==== Year %s | CPT %s (%s) ====", year, cpt_code, cpt_label))

  possible_filters <- list(
    list("filter[HCPCS_CD]" = cpt_code),
    list("filter[HCPCS_Cd]" = cpt_code),
    list("filter[Rndrng_Prvdr_HCPCS_Cd]" = cpt_code)
  )

  year_df <- tibble()

  for (i in seq_along(possible_filters)) {
    tryCatch({
      message(sprintf("    Trying filter variant %d...", i))
      year_df <- fetch_all_for_filter(uuid, possible_filters[[i]], cpt_code)
      if (nrow(year_df) > 0) {
        message(sprintf("    ✓ Success with filter variant %d! Retrieved %d records", i, nrow(year_df)))
        break
      }
    }, error = function(e) {
      message(sprintf("    Filter variant %d failed: %s", i, conditionMessage(e)))
    })
  }

  if (nrow(year_df) == 0) {
    message("  ⚠ No records retrieved for this year")
    return(tibble())
  }

  year_df <- dedupe_year(year_df)
  year_df$Year <- year

  message(sprintf("  Retrieved %d unique records for year %s", nrow(year_df), year))
  return(year_df)
}

# ------------------------ QUERY ALL YEARS/CODES --------------------------------
all_cpt_data <- list()

for (i in seq_len(nrow(cpt_catalog))) {
  cpt_code <- cpt_catalog$CPT_Code[i]
  cpt_label <- cpt_catalog$CPT_Label[i]

  message(sprintf("\n========= Pulling CPT %s (%s) =========", cpt_code, cpt_label))
  annual_data <- list()

  for (yr in names(year_uuid_map)) {
    uuid <- year_uuid_map[[yr]]
    if (is.null(uuid) || is.na(uuid) || uuid == "") {
      message(sprintf("Skipping year %s (UUID missing)", yr))
      next
    }

    yr_data <- tryCatch(
      get_year_data(yr, uuid, cpt_code, cpt_label),
      error = function(e) {
        message(sprintf("  ERROR in year %s: %s", yr, conditionMessage(e)))
        tibble()
      }
    )
    annual_data[[yr]] <- yr_data
  }

  cpt_data <- bind_rows(annual_data)
  if (nrow(cpt_data) == 0) {
    message(sprintf("No data for CPT %s (%s)", cpt_code, cpt_label))
    next
  }

  cpt_data$CPT_Code <- cpt_code
  cpt_data$CPT_Label <- cpt_label
  all_cpt_data[[cpt_code]] <- cpt_data
}

combined_data <- bind_rows(all_cpt_data)

cat(sprintf("Total records retrieved: %d\n", nrow(combined_data)))

if (nrow(combined_data) == 0) {
  stop("No data retrieved. Check if the API endpoint or filter format is correct.")
}

cat("Available columns:\n")
print(names(combined_data))
cat("\n")

service_candidates <- c("PSPS_SUBMITTED_SERVICE_CNT", grep("service", names(combined_data), ignore.case = TRUE, value = TRUE))
payment_candidates <- c("PSPS_NCH_PAYMENT_AMT", grep("payment|pymt", names(combined_data), ignore.case = TRUE, value = TRUE))

services_col <- intersect(service_candidates, names(combined_data))[1]
payment_col <- intersect(payment_candidates, names(combined_data))[1]

if (is.na(services_col)) {
  stop("Could not identify a services column.")
}

cat(sprintf("Using services column: %s\n", services_col))
if (!is.na(payment_col)) {
  cat(sprintf("Using payment column: %s\n", payment_col))
} else {
  cat("No payment column detected; totals will only include service counts.\n")
}
cat("\n")

latera_data <- combined_data %>%
  rename(
    Services = !!services_col,
    Payment = !!payment_col
  ) %>%
  mutate(
    Services = as.numeric(Services),
    Payment = as.numeric(Payment),
    Year = as.numeric(Year)
  )

# CPI-U annual averages (BLS) for 2014-2024; 2024 value uses the latest available
# year-to-date average to express all dollars in 2024 terms.
cpi_table <- tribble(
  ~Year, ~CPI,
  2014, 236.736,
  2015, 237.017,
  2016, 240.007,
  2017, 245.120,
  2018, 251.107,
  2019, 255.657,
  2020, 258.811,
  2021, 270.970,
  2022, 292.655,
  2023, 305.710,
  2024, 314.000
)

cpi_2024 <- cpi_table %>% filter(Year == 2024) %>% pull(CPI)

summary_by_year_cpt <- latera_data %>%
  group_by(Year, CPT_Code, CPT_Label) %>%
  summarise(
    Total_Services = sum(Services, na.rm = TRUE),
    Total_Payments = if ("Payment" %in% names(.)) sum(Payment, na.rm = TRUE) else NA_real_,
    .groups = "drop"
  ) %>%
  right_join(cpt_catalog %>% mutate(Year = list(names(year_uuid_map))) %>% unnest(Year),
             by = c("CPT_Code", "CPT_Label", "Year")) %>%
  mutate(
    Year = as.numeric(Year),
    Total_Services = replace_na(Total_Services, 0),
    Total_Payments = replace_na(Total_Payments, 0)
  ) %>%
  arrange(Year, CPT_Code)

summary_by_year_cpt <- summary_by_year_cpt %>%
  left_join(cpi_table, by = "Year") %>%
  mutate(
    Inflation_Factor_2024 = cpi_2024 / CPI,
    Total_Payments_2024 = Total_Payments * Inflation_Factor_2024
  ) %>%
  select(-CPI)

overall_summary <- summary_by_year_cpt %>%
  group_by(Year) %>%
  summarise(
    Total_Services = sum(Total_Services, na.rm = TRUE),
    Total_Payments = sum(Total_Payments, na.rm = TRUE),
    Total_Payments_2024 = sum(Total_Payments_2024, na.rm = TRUE),
    .groups = "drop"
  )

cat("Summary (by CPT):\n")
print(summary_by_year_cpt)
cat("\nOverall summary (all CPTs):\n")
print(overall_summary)

if ("Total_Payments" %in% names(summary_by_year_cpt) && any(!is.na(summary_by_year_cpt$Total_Payments))) {
  dir.create("plots", showWarnings = FALSE)
  stacked_plot <- ggplot(summary_by_year_cpt, aes(x = factor(Year), y = Total_Payments, fill = CPT_Label)) +

    ) +
    theme_minimal(base_size = 12) +
    theme(legend.position = "bottom")

  adj_plot <- ggplot(summary_by_year_cpt, aes(x = factor(Year), y = Total_Payments_2024, fill = CPT_Label)) +
    geom_col(color = "grey20") +
    labs(
      title = "Inflation-Adjusted PSPS Medicare Payments (2014-2024)",
      subtitle = "Nominal PSPS_NCH_PAYMENT_AMT converted to 2024 dollars using CPI-U",
      x = "Year",
      y = "Total Payments (2024 $)",
      fill = "CPT"
    ) +
    theme_minimal(base_size = 12) +
    theme(legend.position = "bottom")

  ggsave("plots/cpt_payments_2014_2024.png", stacked_plot, width = 10, height = 6, dpi = 320)
  ggsave("plots/cpt_payments_2014_2024_inflation_adjusted.png", adj_plot, width = 10, height = 6, dpi = 320)
  cat("\nStacked bar charts saved to: \n  - plots/cpt_payments_2014_2024.png (nominal)\n  - plots/cpt_payments_2014_2024_inflation_adjusted.png (2024 dollars)\n")
} else {
  cat("\nSkipped plotting because payment totals were unavailable.\n")
}
